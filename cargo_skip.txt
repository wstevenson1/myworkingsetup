# Crates in rust_packages.txt that are excluded from the cargo install loop
# in roles/language_packages/. Without this each is re-attempted on every run:
# a fresh source build that either fails minutes in or rebuilds something the
# host already has another way. One name per line; "#" starts a comment.

# --- no reproducible source ---
# `cargo install --list` recorded the path to a local checkout that only
# exists on the Mac these were captured from; no crates.io or git source.
dusage
dwatch
valerter

# --- macOS only ---
# Fails on Linux with "macpow supports macOS targets only" after a few
# minutes of compiling (uses AF_LINK, Apple `framework` link kind).
macpow

# --- provided by Homebrew instead ---
# homebrew_formulae.txt installs mise earlier in the same play. The cargo
# check only consults `cargo install --list`, so without this entry it
# rebuilds mise from source every run -- a very large dependency tree
# (rustls/aws-lc-rs, gix, rkyv, zstd-sys, sigstore, ...).
mise

# --- does not build here ---
# `palette` (a dependency) fails to compile, ~34 errors, aborting eza. There
# is an `eza` Homebrew formula if it is wanted on the target.
eza
